# RA-34 Manage DB Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Runs and run types live in Supabase Postgres (RLS per user), with a local SQLite copy + outbox so the app works offline, and every screen reads real data instead of `mockData.ts`.

**Architecture:** Cloud schema in `supabase/schema.sql` (Bar runs it once). Phone: `src/db/` (SQLite tables + outbox, pure merge logic, sync orchestrator) and `src/api/sync.ts` (the only Supabase data calls). Zustand stores (`useRunStore`, `useRunTypeStore`) load from SQLite; the auth store drives the lifecycle (load on login, sync on foreground, wipe on logout).

**Tech Stack:** Expo 58, expo-sqlite 58 (sync API), @supabase/supabase-js 2.119, Zustand 5, TypeScript. Self-checks run with plain `node` (Node 24 strips types).

**Spec:** `docs/superpowers/specs/2026-11-18-manage-db-design.md`

**Project rules for the implementer:**
- **Never run git commands.** At the end of each task, stop and list changed files + a suggested commit message for Bar.
- Only `src/api/` may import `@supabase/supabase-js` or `src/api/supabase.ts`.
- HealthKit data is never changed by edits; edits live in `runs.edits`.
- Verify types with `npx tsc --noEmit` (must print nothing).

**Deviations from the spec (decided while planning, small):**
- No `useUserStore` / local `profiles` table yet: nothing can change `unit` until RA-6, and screens still hardcode `UNIT = 'km'`. The cloud `profiles` table + trigger are still created (RA-34 criterion).
- No `setType` / `setNotes` actions: no screen sets a type or writes a note yet. The only user write today is Edit Run → `editRun`. Device test 2 therefore tests offline *edits* only.
- The seed lives as `seedSampleRuns()` in `useRunStore` (mapping logic in `src/db/merge.ts`), not a separate `seed.ts`.

create table public.profiles (
  user_id    uuid primary key default auth.uid() references auth.users on delete cascade,
  unit       text not null default 'km' check (unit in ('km', 'mi')),
  updated_at timestamptz not null default now()
);

create table public.run_types (
  id         uuid primary key default gen_random_uuid(),
  user_id    uuid not null default auth.uid() references auth.users on delete cascade,
  name       text not null,
  hue        int  not null,
  deleted    boolean not null default false,
  updated_at timestamptz not null default now()
);

create table public.runs (
  id             uuid primary key,               -- = HealthKit workout UUID for Apple Health runs
  user_id        uuid not null default auth.uid() references auth.users on delete cascade,
  healthkit_uuid text,
  type_id        uuid references public.run_types on delete set null,  -- null = unclassified
  name           text not null,
  date           date not null,
  distance_km    real not null,
  time_sec       int  not null,
  avg_hr         int,
  max_hr         int,
  min_hr         int,
  elev_gain      int,
  cadence        int,
  source         text not null,
  notes          text,
  splits         jsonb not null,
  segments       jsonb,
  zones          jsonb,
  weather        jsonb,
  edits          jsonb not null default '{}',    -- manual edits on top of HealthKit values
  updated_at     timestamptz not null default now(),
  unique (user_id, healthkit_uuid)
);
create index runs_user_updated on public.runs (user_id, updated_at);
create index run_types_user_updated on public.run_types (user_id, updated_at);

grant select, insert, update, delete on public.profiles, public.run_types, public.runs to authenticated;

-- ─── Row Level Security: each user sees and changes only their own rows ─────

alter table public.profiles  enable row level security;
alter table public.run_types enable row level security;
alter table public.runs      enable row level security;

create policy "own rows" on public.profiles  for all to authenticated
  using (user_id = (select auth.uid())) with check (user_id = (select auth.uid()));
create policy "own rows" on public.run_types for all to authenticated
  using (user_id = (select auth.uid())) with check (user_id = (select auth.uid()));
create policy "own rows" on public.runs      for all to authenticated
  using (user_id = (select auth.uid())) with check (user_id = (select auth.uid()));

language plpgsql set search_path = '' as $$
begin
  new.updated_at = now();
  return new;
end $$;

create trigger touch before insert or update on public.profiles  for each row execute function public.touch_updated_at();
create trigger touch before insert or update on public.run_types for each row execute function public.touch_updated_at();
create trigger touch before insert or update on public.runs      for each row execute function public.touch_updated_at();

begin
  insert into public.profiles (user_id) values (new.id);
  insert into public.run_types (user_id, name, hue) values
    (new.id, 'Race', 20), (new.id, 'Tempo', 70), (new.id, 'Hills', 120), (new.id, 'Easy', 170),

  for each row execute function public.handle_new_user();

from auth.users u
cross join (values ('Race', 20), ('Tempo', 70), ('Hills', 120), ('Easy', 170),

// Pure data logic for the DB layer (no React Native imports, so `node src/db/sync.check.ts` can run it).
import type { Run, RunEdits, RunType } from '../types/index.ts'

/** Columns holding JSON: TEXT in SQLite, jsonb in Postgres. */
export const JSON_COLS = ['splits', 'segments', 'zones', 'weather', 'edits'] as const

/** A run as stored (same snake_case columns locally and in the cloud), JSON columns parsed. */
export interface RunRow {
  id: string
  healthkit_uuid: string | null
  type_id: string | null
  name: string
  date: string
  distance_km: number
  time_sec: number
  avg_hr: number | null
  max_hr: number | null
  min_hr: number | null
  elev_gain: number | null
  cadence: number | null
  source: Run['source']
  notes: string | null
  splits: Run['splits']
  segments: Run['segments']
  zones: Run['zones']
  weather: Run['weather']
  edits: RunEdits
  updated_at?: string | null
}

/** Stored row → what screens show: HealthKit values with the user's edits on top. */
export function rowToRun(r: RunRow): Run {
  return {
    id: r.id, healthkitUuid: r.healthkit_uuid ?? '', typeId: r.type_id, name: r.name, date: r.date,
    distanceKm: r.distance_km, timeSec: r.time_sec, avgHr: r.avg_hr, maxHr: r.max_hr, minHr: r.min_hr,
    elevGain: r.elev_gain, cadence: r.cadence, source: r.source, notes: r.notes,
    splits: r.splits, segments: r.segments, zones: r.zones, weather: r.weather,
    ...r.edits,
    edits: r.edits,
  }
}

/** Run (without edits) → row. Used for imports (seed now, HealthKit in RA-3). */
export function runToRow(r: Run): RunRow {
  return {
    id: r.id, healthkit_uuid: r.healthkitUuid || null, type_id: r.typeId, name: r.name, date: r.date,
    distance_km: r.distanceKm, time_sec: r.timeSec, avg_hr: r.avgHr, max_hr: r.maxHr, min_hr: r.minHr,
    elev_gain: r.elevGain, cadence: r.cadence, source: r.source, notes: r.notes,
    splits: r.splits, segments: r.segments, zones: r.zones, weather: r.weather,
    edits: {},
  }
}

/** New edits after saving `patch`: a field equal to the HealthKit value is dropped (= not edited). */
export function diffEdits(raw: Run, current: RunEdits, patch: RunEdits): RunEdits {
  const next: Record<string, unknown> = { ...current }
  for (const [k, v] of Object.entries(patch)) {
    if (v === raw[k as keyof Run]) delete next[k]
    else next[k] = v
  }
  return next as RunEdits
}

/** Pulled rows minus those with unsent local changes (the local change wins until it is pushed). */
export function filterPulled<T extends { id: string }>(rows: T[], pendingIds: Set<string>): T[] {
  return rows.filter(r => !pendingIds.has(r.id))
}

// ─── Dev seed ────────────────────────────────────────────────────────────────

const MOCK_TYPE_NAMES: Record<string, string> = { easy: 'Easy', long: 'Long', tempo: 'Tempo', interval: 'Interval' }

/** Mock healthkitUuid 'hk-<uuid>' → '<uuid>': the same id on every press and every device. */
export const seedUuid = (hk: string) => hk.replace(/^hk-/, '')

export function seedRows(mock: Run[], types: RunType[]): RunRow[] {
  const byName = Object.fromEntries(types.map(t => [t.name, t.id]))
  return mock.map(r => {
    const id = seedUuid(r.healthkitUuid)
    const typeId = r.typeId ? byName[MOCK_TYPE_NAMES[r.typeId]] ?? null : null
    return runToRow({ ...r, id, healthkitUuid: id, typeId })
  })
}

```ts
// All local reads/writes. Every user change goes through writeChange (local row + outbox, one transaction).
import type { SQLiteBindValue } from 'expo-sqlite'
import { db } from './schema'
import { JSON_COLS, filterPulled, type RunRow } from './merge'

export type Table = 'runs' | 'run_types'
export interface RunTypeRow { id: string; name: string; hue: number; deleted: number }
export interface PendingOp { seq: number; tbl: Table; row_id: string; kind: 'insert' | 'update'; patch: string }

const COLS: Record<Table, readonly string[]> = {
  runs: ['id', 'healthkit_uuid', 'type_id', 'name', 'date', 'distance_km', 'time_sec', 'avg_hr', 'max_hr',
    'min_hr', 'elev_gain', 'cadence', 'source', 'notes', 'splits', 'segments', 'zones', 'weather', 'edits', 'updated_at'],
  run_types: ['id', 'name', 'hue', 'deleted', 'updated_at'],
}

// JS value → SQLite value: objects/arrays as JSON text, booleans as 0/1
function toSql(v: unknown): SQLiteBindValue {
  if (v === undefined || v === null) return null
  if (typeof v === 'boolean') return v ? 1 : 0
  if (typeof v === 'object') return JSON.stringify(v)
  return v as SQLiteBindValue
}

function parseRun(r: Record<string, unknown>): RunRow {
  const out: Record<string, unknown> = { ...r }
  for (const c of JSON_COLS) out[c] = r[c] == null ? (c === 'edits' ? {} : null) : JSON.parse(r[c] as string)
  return out as unknown as RunRow
}

// ─── Reads ───────────────────────────────────────────────────────────────────

export const loadRuns = (): RunRow[] =>
  db.getAllSync<Record<string, unknown>>('SELECT * FROM runs ORDER BY date DESC').map(parseRun)

export function getRunRow(id: string): RunRow | null {
  const r = db.getFirstSync<Record<string, unknown>>('SELECT * FROM runs WHERE id = ?', [id])
  return r ? parseRun(r) : null
}

export const loadRunTypes = (): RunTypeRow[] =>
  db.getAllSync<RunTypeRow>('SELECT * FROM run_types WHERE deleted = 0 ORDER BY hue')

// ─── Writes ──────────────────────────────────────────────────────────────────

function upsertLocal(tbl: Table, row: Record<string, unknown>) {
  const cols = COLS[tbl].filter(c => c in row)  // drops cloud-only columns such as user_id
  db.runSync(
    `INSERT INTO ${tbl} (${cols.join(',')}) VALUES (${cols.map(() => '?').join(',')})
     ON CONFLICT(id) DO UPDATE SET ${cols.filter(c => c !== 'id').map(c => `${c} = excluded.${c}`).join(',')}`,
    cols.map(c => toSql(row[c])),
  )
}

/**
 * A user change: update the local row and queue it for upload, atomically.
 * insert = add if missing (an existing row is left alone and nothing is queued);
 * update = set only the given columns.
 */
export function writeChange(tbl: Table, kind: 'insert' | 'update', row: { id: string } & Record<string, unknown>) {
  db.withTransactionSync(() => {
    if (kind === 'insert') {
      const cols = COLS[tbl].filter(c => c in row)
      const res = db.runSync(
        `INSERT OR IGNORE INTO ${tbl} (${cols.join(',')}) VALUES (${cols.map(() => '?').join(',')})`,
        cols.map(c => toSql(row[c])),
      )
      if (res.changes === 0) return
    } else {
      const cols = COLS[tbl].filter(c => c !== 'id' && c in row)
      db.runSync(`UPDATE ${tbl} SET ${cols.map(c => `${c} = ?`).join(',')} WHERE id = ?`,
        [...cols.map(c => toSql(row[c])), row.id])
    }
    db.runSync('INSERT INTO pending_ops (tbl, row_id, kind, patch) VALUES (?, ?, ?, ?)',
      [tbl, row.id, kind, JSON.stringify(row)])
  })
}

/** Rows from the cloud. Rows with unsent local changes are skipped (local wins until pushed). */
export function applyPulled(tbl: Table, rows: Record<string, unknown>[]) {
  const pending = new Set(pendingOps().map(o => o.row_id))
  db.withTransactionSync(() => {
    for (const r of filterPulled(rows as ({ id: string } & Record<string, unknown>)[], pending)) upsertLocal(tbl, r)
  })
}

// ─── Outbox ──────────────────────────────────────────────────────────────────

/** Ops still to send, oldest first. Failed ops (rejected by the server) are kept but skipped. */
export const pendingOps = (): PendingOp[] =>
  db.getAllSync<PendingOp>('SELECT seq, tbl, row_id, kind, patch FROM pending_ops WHERE failed = 0 ORDER BY seq')
export const pendingCount = () =>
  db.getFirstSync<{ n: number }>('SELECT COUNT(*) AS n FROM pending_ops WHERE failed = 0')?.n ?? 0
export const deleteOp = (seq: number) => db.runSync('DELETE FROM pending_ops WHERE seq = ?', [seq])
export const markOpFailed = (seq: number) => db.runSync('UPDATE pending_ops SET failed = 1 WHERE seq = ?', [seq])

// ─── Meta (last pull time per table, owner of the local data) ────────────────

export const getMeta = (key: string) =>
  db.getFirstSync<{ value: string }>('SELECT value FROM sync_state WHERE key = ?', [key])?.value ?? null
export const setMeta = (key: string, value: string) =>
  db.runSync('INSERT OR REPLACE INTO sync_state (key, value) VALUES (?, ?)', [key, value])

/** Log out: the next user of this phone must see nothing of this one. */
export function clearAll() {
  db.execSync('DELETE FROM runs; DELETE FROM run_types; DELETE FROM pending_ops; DELETE FROM sync_state;')
}
```

```ts
// The only code that sends data rows to / reads them from Supabase. src/db/sync.ts drives it.
import { supabase } from './supabase'

export type PushResult = 'ok' | 'retry' | 'failed'

/** Send one outbox op. 'retry' = try again later (offline, expired session, server hiccup). */
export async function pushOp(
  tbl: string, kind: 'insert' | 'update', rowId: string, patch: Record<string, unknown>,
): Promise<PushResult> {
  const { updated_at: _ignored, ...row } = patch  // the server stamps updated_at itself
  try {
    const { error, status } = kind === 'insert'
      // ignoreDuplicates: a run already in the cloud is left as is (never overwrites type/notes/edits)
      ? await supabase.from(tbl).upsert(row, { onConflict: 'id', ignoreDuplicates: true })
      : await supabase.from(tbl).update(row).eq('id', rowId)
    if (!error) return 'ok'
    if (status === 0 || status === 401 || status === 408 || status === 429 || status >= 500) return 'retry'
    console.warn('[sync] rejected', tbl, kind, rowId, status, error.message)
    return 'failed'
  } catch {
    return 'retry'
  }
}

const PAGE = 1000  // Supabase returns at most 1000 rows per request

/** Rows changed after `since` (all rows when null), oldest first. null = request failed. */
export async function pullSince(tbl: string, since: string | null): Promise<Record<string, unknown>[] | null> {
  const rows: Record<string, unknown>[] = []
  try {
    for (let from = 0; ; from += PAGE) {
      let q = supabase.from(tbl).select('*').order('updated_at').order('id').range(from, from + PAGE - 1)
      if (since) q = q.gt('updated_at', since)
      const { data, error } = await q
      if (error) return null
      rows.push(...data)
      if (data.length < PAGE) return rows
    }
  } catch {
    return null
  }
}
```

```ts
// syncNow(): push the outbox, then pull what changed in the cloud. Safe to call any time, any number of times.
import { pullSince, pushOp } from '../api/sync'
import { applyPulled, deleteOp, getMeta, markOpFailed, pendingOps, setMeta, type Table } from './queries'

const TABLES: Table[] = ['run_types', 'runs']  // types first: runs point at them

const listeners = new Set<() => void>()
/** Called after a pull changed local data (stores reload). */
export function onPulled(cb: () => void) {
  listeners.add(cb)
  return () => { listeners.delete(cb) }
}

let running: Promise<void> | null = null
let again = false

/** Single-flight: a call during a sync makes it run once more afterwards instead of in parallel. */
export function syncNow(): Promise<void> {
  if (running) { again = true; return running }
  running = (async () => {
    do { again = false; await syncOnce() } while (again)
  })().finally(() => { running = null })
  return running
}

// ponytail: a logout during a sync can let a pull land after clearAll(); the owner check on next login wipes it.
async function syncOnce() {
  for (const op of pendingOps()) {
    const res = await pushOp(op.tbl, op.kind, op.row_id, JSON.parse(op.patch))
    if (res === 'retry') return  // offline or temporary: keep this op and the rest for next time
    if (res === 'ok') deleteOp(op.seq)
    else markOpFailed(op.seq)
  }

  let changed = false
  for (const tbl of TABLES) {
    const key = `pulled:${tbl}`
    const rows = await pullSince(tbl, getMeta(key))
    if (!rows) return
    if (rows.length === 0) continue
    applyPulled(tbl, rows)
    setMeta(key, String(rows[rows.length - 1].updated_at))  // rows are ordered by updated_at
    changed = true
  }
  if (changed) listeners.forEach(cb => cb())
}
```
