> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the mock login/signup with real Supabase accounts: email+password with confirmation, password reset, persistent session, logout, account deletion, and native Google + Apple sign-in.

**Architecture:** One Supabase client in `src/api/supabase.ts`; every auth call goes through `src/api/auth.ts`, which returns `null` on success or a user-facing message string. - [ ] **Step 6: Implement `src/api/authErrors.ts`**

```ts
// Maps Supabase auth errors to the messages shown on screen.
// Codes: https://supabase.com/docs/guides/auth/debugging/error-codes
export type AuthErrorLike = { code?: string; name?: string; message?: string }

export function authErrorMessage(e: AuthErrorLike): string {
  if (e.name === 'AuthRetryableFetchError' || /network request failed/i.test(e.message ?? '')) {
    return 'No connection, try again'
  }
  switch (e.code) {
    case 'invalid_credentials':        return 'Email or password is incorrect'
    case 'email_not_confirmed':        return 'Please confirm your email first'
    case 'user_already_exists':
    case 'email_exists':               return 'This email already has an account. Log in?'
    case 'weak_password':              return "Password doesn't meet the requirements"
    case 'over_email_send_rate_limit':
    case 'over_request_rate_limit':    return 'Too many attempts. Wait a minute and try again'
    case 'otp_expired':
    case 'flow_state_expired':
    case 'flow_state_not_found':
    case 'bad_code_verifier':          return 'Link expired. Request a new one'
    case 'same_password':              return 'New password must be different from the old one'
  }
  console.warn('Unhandled auth error', e)
  return 'Something went wrong, try again'
}
```

- [ ] **Step 9: Create `src/api/auth.ts` (session + sign-out part)**

```ts
// All auth operations. Each returns null on success or a message to show the user.
import type { Session } from '@supabase/supabase-js'
import { supabase } from './supabase'
import { authErrorMessage, type AuthErrorLike } from './authErrors'

export type { Session }
type Result = Promise<string | null>

async function run(op: () => Promise<{ error: unknown }>): Result {
  try {
    const { error } = await op()
    return error ? authErrorMessage(error as AuthErrorLike) : null
  } catch (e) {
    return authErrorMessage(e as AuthErrorLike)
  }
}

/** Calls cb with the current session immediately (INITIAL_SESSION) and on every change. */
export function onSessionChange(cb: (session: Session | null) => void): () => void {
  const { data } = supabase.auth.onAuthStateChange((_event, session) => cb(session))
  return () => data.subscription.unsubscribe()
}

export const signOut = (): Result => run(() => supabase.auth.signOut())
```

```ts
// Expo Go → exp://<ip>:8081/--/<path>; dev/prod build → chronodrom://<path>
const linkTo = (path: string) => Linking.createURL(path)

export async function signUp(name: string, email: string, password: string): Result {
  try {
    const { data, error } = await supabase.auth.signUp({
      email,
      password,
      options: { data: { name }, emailRedirectTo: linkTo('auth-callback') },
    })
    if (error) return authErrorMessage(error)
    // With email confirmation on, an existing email returns a user with no identities (no error).
    if (data.user && data.user.identities?.length === 0) return authErrorMessage({ code: 'user_already_exists' })
    return null
  } catch (e) {
    return authErrorMessage(e as AuthErrorLike)
  }
}

export const signIn = (email: string, password: string): Result =>
  run(() => supabase.auth.signInWithPassword({ email, password }))

export const resendConfirmation = (email: string): Result =>
  run(() => supabase.auth.resend({ type: 'signup', email, options: { emailRedirectTo: linkTo('auth-callback') } }))

/** Exchanges the ?code= from an email link for a session (PKCE). */
export const exchangeAuthCode = (code: string): Result =>
  run(() => supabase.auth.exchangeCodeForSession(code))
```

```tsx
import { useEffect, useRef, useState } from 'react'

import { router, useLocalSearchParams } from 'expo-router'
import { colors } from '../src/utils/tokens'
import { exchangeAuthCode } from '../src/api/auth'

// Opened by the email-confirmation link: chronodrom://auth-callback?code=…
export default function AuthCallback() {
  const { code } = useLocalSearchParams<{ code?: string }>()
  const [error, setError] = useState<string | null>(null)
  const started = useRef(false)  // the code is single-use; never exchange twice

  useEffect(() => {
    if (started.current) return
    started.current = true
    if (!code) { setError('Link expired. Request a new one'); return }
    exchangeAuthCode(code).then(err => (err ? setError(err) : router.replace('/')))

    <View style={s.screen}>

          <Text style={s.error}>{error}</Text>
          <TouchableOpacity onPress={() => router.replace('/')} activeOpacity={0.7}>

          </TouchableOpacity>
        </>

      )}

  )
}
```

```tsx
import { useEffect, useRef, useState } from 'react'
import { View, Text, TextInput, TouchableOpacity, ActivityIndicator, StyleSheet } from 'react-native'
import { router, useLocalSearchParams } from 'expo-router'
import { colors } from '../src/utils/tokens'
import { exchangeAuthCode, updatePassword } from '../src/api/auth'
import { PASSWORD_RULES, passwordValid } from '../src/utils/password'

// Opened by the reset email link: chronodrom://reset-password?code=…
// Always reachable in the gate, so it is not redirected away when the recovery session appears.
export default function ResetPassword() {
  const { code } = useLocalSearchParams<{ code?: string }>()
  const [ready,    setReady]    = useState(false)
  const [password, setPassword] = useState('')
  const [confirm,  setConfirm]  = useState('')
  const [busy,     setBusy]     = useState(false)

    setBusy(false)
    if (err) setError(err)
    else router.replace('/')  // logged in → dashboard
  }

  if (!ready) {
    return (
      <View style={[s.screen, { alignItems: 'center' }]}>
        {error ? (
          <>
            <Text style={s.error}>{error}</Text>
            <TouchableOpacity onPress={() => router.replace('/')} activeOpacity={0.7}>

          </>

    )
```

```ts
export async function signInWithApple(): Result {
  try {
    const cred = await AppleAuthentication.signInAsync({
      requestedScopes: [
        AppleAuthentication.AppleAuthenticationScope.FULL_NAME,
        AppleAuthentication.AppleAuthenticationScope.EMAIL,
      ],
    })
    if (!cred.identityToken) return authErrorMessage({ code: 'no_identity_token' })
    const { error } = await supabase.auth.signInWithIdToken({ provider: 'apple', token: cred.identityToken })
    if (error) return authErrorMessage(error)
    // Apple sends the name only on the very first sign-in — save it now or never.
    const name = [cred.fullName?.givenName, cred.fullName?.familyName].filter(Boolean).join(' ')
    if (name) await supabase.auth.updateUser({ data: { name } })
    return null
  } catch (e) {
    if ((e as { code?: string }).code === 'ERR_REQUEST_CANCELED') return null  // user closed the sheet
    return authErrorMessage(e as AuthErrorLike)
  }
}

export async function signInWithGoogle(): Result {
  try {
    // Loaded lazily: the native module doesn't exist in Expo Go, and a top-level import would crash it.
    const { GoogleSignin, isSuccessResponse } = await import('@react-native-google-signin/google-signin')
    GoogleSignin.configure({
      webClientId: process.env.EXPO_PUBLIC_GOOGLE_WEB_CLIENT_ID,
      iosClientId: process.env.EXPO_PUBLIC_GOOGLE_IOS_CLIENT_ID,
    })
    const res = await GoogleSignin.signIn()
    if (!isSuccessResponse(res)) return null  // cancelled
    if (!res.data.idToken) return authErrorMessage({ code: 'no_identity_token' })
    return run(() => supabase.auth.signInWithIdToken({ provider: 'google', token: res.data.idToken! }))
  } catch (e) {
    return authErrorMessage(e as AuthErrorLike)
  }
}
```
