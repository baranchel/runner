# RA-39 Set Run Type Manually — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
>
> **Project rules:** never run git commands (Bar commits). Subagents: Sonnet minimum, never Haiku. Read `CLAUDE.md` first. Windows + Git Bash; run checks from the repo root.

**Goal:** The user can set a run's type from a pill on Run Detail, new runs (started after signup) must be answered via a Dashboard card, and many runs can be set at once from All runs.

**Architecture:** One new column `runs.needs_type` (cloud + SQLite v7). One store action `setType(ids, typeId)` writes through the existing outbox (`writeChange` → `syncNow`). One picker function `pickRunType()` on top of the existing native action sheet `pick()`. Screens only call those two.

**Tech Stack:** Expo / React Native, expo-router, Zustand, expo-sqlite, Supabase. Self-checks run with plain `node` (no test framework).

**Spec:** `docs/superpowers/specs/2026-11-22-run-type-design.md`

export function pickType(onPick: (typeId: string | null) => void) {
  const types = useRunTypeStore.getState().types
  pick('Run type', ['All types', 'Unclassified', ...types.map(t => t.name)],
    i => onPick(i === 0 ? null : i === 1 ? UNCLASSIFIED : types[i - 2].id))
}

export function typeLabel(typeId: string | null) {
  if (!typeId) return 'All types'
  if (typeId === UNCLASSIFIED) return 'Unclassified'
  return useRunTypeStore.getState().typeMap[typeId]?.name ?? 'All types'
}

/** RA-39 run-type picker (pill, Dashboard card, bulk; RA-15 "Change"). current undefined = no ✓ (bulk). */
export function pickRunType(title: string, current: string | null | undefined, onPick: (typeId: string | null) => void) {
  const types = useRunTypeStore.getState().types
  const ids: (string | null)[] = [null, ...types.map(t => t.id)]
  const names = ['Unclassified', ...types.map(t => t.name)]
  pick(title, names.map((n, i) => (ids[i] === current ? `✓ ${n}` : n)), i => onPick(ids[i]))
}
