# RA-8 Shoes Collection Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Track each pair of running shoes, its distance against a limit, a default shoe per run type, and warn at 90% / 100%.

**Architecture:** Shoes are a new per-user table that goes through the existing outbox sync (`writeChange` → `syncNow`), like runs and profiles. `runs.shoe_id` links a run to a shoe; `run_types.default_shoe_id` holds each type's default. Distance, levels and alerts are pure functions in `src/shoes/shoes.ts` (self-check). The brand/model catalogue is two read-only Supabase tables fetched only on Add shoe; logos come from a public Storage bucket by derived file name.

**Tech Stack:** Expo / React Native, expo-router, Zustand, expo-sqlite, Supabase (Postgres + Storage), expo-crypto (`randomUUID`), `@react-native-community/datetimepicker`.

**Spec:** `docs/superpowers/specs/2026-11-23-shoes-design.md`. Branch `feature/shoes` (stacked on `feature/run-type`).

**Project rules for implementers:**
- **Never run git.** Bar commits. Each task ends with "Stop: list changed files" instead of a commit.
- Only `src/api/` imports Supabase. Pure files (`src/shoes/shoes.ts`, `src/db/merge.ts`) import nothing from React Native and use `.ts` extensions in relative imports so `node <file>.check.ts` runs them.
- Distances in km only (`UNIT = 'km'`), JetBrains Mono for numbers (`fonts.mono`), colours from `src/utils/tokens.ts`.
- Verification commands: `npx tsc --noEmit`, `node src/shoes/shoes.check.ts`, `node src/db/sync.check.ts`, `node src/healthkit/import.check.ts`, `npx tsx src/utils/compare.check.ts`.

```ts
// Pure shoe logic (RA-8). No React Native imports: `node src/shoes/shoes.check.ts` runs it.
// Distance and levels are never stored: they follow the runs, like HR zones follow hr_hist.
import type { Run } from '../types/index.ts'

export const COLOR_KEYS = ['black', 'white', 'grey', 'red', 'orange', 'yellow', 'green', 'blue', 'purple', 'pink', 'multi'] as const
export type ColorKey = typeof COLOR_KEYS[number]

/** A shoe as stored (same columns in SQLite and the cloud; user_id is cloud only). */
export interface ShoeRow {
  id: string
  brand: string
  model: string
  name: string | null
  color: ColorKey | null
  in_use_from: string       // 'YYYY-MM-DD'
  limit_km: number
  extra_km: number          // km not covered by assigned runs
  retired: boolean
  alert_level: number       // highest pop-up already shown: 0 | 90 | 100
  card_dismissed: number    // highest Dashboard card dismissed: 0 | 90 | 100
  updated_at?: string | null
}

export type Level = 0 | 90 | 100
export const GENERIC_KM = 700  // limit for a model typed with "Other…"

/** 100 at/over the limit, 90 from 90%, else 0. Integer maths: 0.9 * 700 is 630.0000000000001. */
export const level = (distKm: number, limitKm: number): Level =>
  limitKm <= 0 ? 0 : distKm >= limitKm ? 100 : distKm * 10 >= limitKm * 9 ? 90 : 0

export const shoeRuns = (shoeId: string, runs: Run[]) => runs.filter(r => r.shoeId === shoeId)

export const shoeDistance = (shoe: ShoeRow, runs: Run[]) =>
  shoeRuns(shoe.id, runs).reduce((s, r) => s + r.distanceKm, 0) + shoe.extra_km

/** Level to pop up now (then store it in alert_level), or null. */
export function popupLevel(shoe: ShoeRow, distKm: number): Level | null {
  if (shoe.retired) return null
  const l = level(distKm, shoe.limit_km)
  return l > shoe.alert_level ? l : null
}

/** Level the Dashboard card shows, or null (dismissing stores it in card_dismissed). */
export function cardLevel(shoe: ShoeRow, distKm: number): Level | null {
  if (shoe.retired) return null
  const l = level(distKm, shoe.limit_km)
  return l > shoe.card_dismissed ? l : null
}

/** After a limit / extra-km edit: remembered levels never stay above the current one, so alerts can fire again. */
export function lowerLevels(shoe: ShoeRow, distKm: number): Pick<ShoeRow, 'alert_level' | 'card_dismissed'> {
  const l = level(distKm, shoe.limit_km)
  return { alert_level: Math.min(shoe.alert_level, l), card_dismissed: Math.min(shoe.card_dismissed, l) }
}

/** Shoe a run gets when its type is set: keeps an existing shoe, else the type's default if it is active. */
export function defaultFill(current: string | null, typeDefault: string | null | undefined, activeIds: Set<string>): string | null {
  if (current) return current
  return typeDefault && activeIds.has(typeDefault) ? typeDefault : null
}

/** Runs the past-runs step offers: on/after "in use from", without a shoe. */
export const pastRunCandidates = (inUseFrom: string, runs: Run[]) =>
  runs.filter(r => !r.shoeId && r.date >= inUseFrom)

/** Extra km from the edited "Distance on this shoe". */
export const extraKm = (typedKm: number, assignedKm: number) => Math.max(0, typedKm - assignedKm)

export function shoeStats(shoe: ShoeRow, runs: Run[]) {
  const rs = shoeRuns(shoe.id, runs)
  const km = rs.reduce((s, r) => s + r.distanceKm, 0)
  const sec = rs.reduce((s, r) => s + r.timeSec, 0)
  return {
    distanceKm: km + shoe.extra_km,
    runCount: rs.length,
    avgPaceSecPerKm: km > 0 ? sec / km : null,
    lastUsed: rs.reduce<string | null>((d, r) => (d === null || r.date > d ? r.date : d), null),
  }
}

const cap = (s: string) => s[0].toUpperCase() + s.slice(1)

export const displayName = (s: Pick<ShoeRow, 'name' | 'model' | 'color'>) =>
  s.name?.trim() || (s.color ? `${s.model} · ${cap(s.color)}` : s.model)

/** Logo file in the brand-logos bucket: "New Balance" → "new-balance.png". */
export const logoFile = (brand: string) =>
  brand.trim().toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '') + '.png'

export const initials = (brand: string) =>
  brand.split(/[\s-]+/).filter(Boolean).slice(0, 2).map(w => w[0].toUpperCase()).join('')
```

```js
// Run: node scripts/gen-shoe-catalogue.mjs
// Source of truth: the appendix table in docs/superpowers/specs/2026-11-23-shoes-design.md
// (columns: Brand | 750 | 700 | 650 | 550 | 400). Re-run after editing the table, then run the SQL again.
import { readFileSync, writeFileSync } from 'node:fs'

const spec = readFileSync(new URL('../docs/superpowers/specs/2026-11-23-shoes-design.md', import.meta.url), 'utf8')
const appendix = spec.slice(spec.indexOf('## Appendix'))
const KM = [750, 700, 650, 550, 400]
const q = s => `'${s.replace(/'/g, "''")}'`

const brands = []
for (const line of appendix.split('\n')) {
  if (!line.startsWith('| ') || line.startsWith('| Brand') || line.startsWith('|---')) continue
  const cells = line.split('|').slice(1, -1).map(c => c.trim())
  const [brand, ...cols] = cells
  const models = cols.flatMap((c, i) => c.split(',').map(m => m.trim()).filter(Boolean).map(m => [m, KM[i]]))
  brands.push({ brand, models })
}
if (brands.length < 20) throw new Error(`only ${brands.length} brands parsed — table format changed?`)

let sql = `-- GENERATED by scripts/gen-shoe-catalogue.mjs from the RA-8 spec appendix. Do not edit by hand.
-- Run in the Supabase SQL Editor after 2026-11-24-shoes.sql. Safe to run again (updates km, adds new rows).

insert into public.shoe_brands (name, sort) values
${brands.map((b, i) => `  (${q(b.brand)}, ${i + 1})`).join(',\n')}
on conflict (name) do update set sort = excluded.sort;

insert into public.shoe_models (brand_id, name, recommended_km)
select b.id, m.name, m.km from (values
${brands.flatMap(b => b.models.map(([m, km]) => `  (${q(b.brand)}, ${q(m)}, ${km})`)).join(',\n')}
) as m(brand, name, km)
join public.shoe_brands b on b.name = m.brand
on conflict (brand_id, name) do update set recommended_km = excluded.recommended_km;
`
writeFileSync(new URL('../supabase/migrations/2026-11-24-shoe-catalogue.sql', import.meta.url), sql)
console.log(`${brands.length} brands, ${brands.reduce((n, b) => n + b.models.length, 0)} models`)
```

```ts
    const name = displayName(shoe)

        { text: 'Retire', onPress: () => retire(shoe.id) },

    return
  }
```

```tsx
// RA-8: after adding a shoe with a past "in use from": tick the runs it was worn on, then confirm

import { useMemo, useState } from 'react'
import { ScrollView, StyleSheet, Text, TextInput, TouchableOpacity, View } from 'react-native'
import { useLocalSearchParams, useRouter } from 'expo-router'
import { useSafeAreaInsets } from 'react-native-safe-area-context'

export default function PastRuns() {
  const router = useRouter()
  const insets = useSafeAreaInsets()
  const { id } = useLocalSearchParams<{ id: string }>()
  const shoe = useShoeStore(s => s.shoes.find(x => x.id === id))
  const runs = useRunStore(s => s.runs)

  const candidates = useMemo(() => (shoe ? pastRunCandidates(shoe.in_use_from, runs) : []), [shoe?.id])

    editShoe(shoe.id, { extra_km: Number.isFinite(km) ? extraKm(km, sum) : 0 })
    useRunStore.getState().setShoe([...ticked], shoe.id)  // also runs the 90% / 100% check

      </View>
```
